{# Jinja macros for mobile-first form controls. Kept deliberately small — each macro = one reusable unit. #} {# segmented: a radio group styled as a segmented control - name: input name (must be unique per form) - options: list of (value, label) tuples, or list of strings - selected: current value (any type; compared as str) #} {% macro segmented(name, options, selected) %}
{% for opt in options %} {% if opt is iterable and opt is not string %} {% set value = opt[0] %} {% set label = opt[1] %} {% else %} {% set value = opt %} {% set label = opt %} {% endif %} {% endfor %}
{% endmacro %} {# field_group: labelled wrapper for inputs/textareas. #} {% macro field_group(label, for_id=None) %} {% endmacro %}
